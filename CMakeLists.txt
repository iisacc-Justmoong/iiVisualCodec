function(visual_library target component)
    add_library(${target} ${ARGN})
    add_library(iiVisualCodec::${component} ALIAS ${target})
    target_compile_features(${target} PUBLIC cxx_std_23)
    target_include_directories(${target} PUBLIC
        $<BUILD_INTERFACE:${CMAKE_CURRENT_BINARY_DIR}/include>
        $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>
        PRIVATE src/Core src/Image)
    set_target_properties(${target} PROPERTIES POSITION_INDEPENDENT_CODE ON CXX_EXTENSIONS OFF EXPORT_NAME ${component} WINDOWS_EXPORT_ALL_SYMBOLS ON)
    if(BUILD_SHARED_LIBS)
        if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
            set_target_properties(${target} PROPERTIES FRAMEWORK TRUE FRAMEWORK_VERSION A
                MACOSX_FRAMEWORK_IDENTIFIER "com.iisacc.${target}")
        elseif(APPLE)
            set_property(TARGET ${target} APPEND PROPERTY INSTALL_RPATH "@loader_path")
        elseif(UNIX AND NOT EMSCRIPTEN)
            set_property(TARGET ${target} APPEND PROPERTY INSTALL_RPATH "$ORIGIN")
        endif()
        if(EMSCRIPTEN)
            target_compile_options(${target} PRIVATE -fPIC -sSIDE_MODULE=1)
            target_link_options(${target} PRIVATE -sSIDE_MODULE=1)
        endif()
    endif()
    if(IIVISUALCODEC_SANITIZE AND CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
        target_compile_options(${target} PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(${target} PUBLIC -fsanitize=address,undefined)
    endif()
    install(TARGETS ${target} EXPORT iiVisualCodec${component}Targets
        ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
        RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR} FRAMEWORK DESTINATION ${CMAKE_INSTALL_LIBDIR})
    install(EXPORT iiVisualCodec${component}Targets NAMESPACE iiVisualCodec:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/iiVisualCodec)
endfunction()
